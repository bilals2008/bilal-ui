---
title: Changelog
description: Latest updates and announcements.
---

import { Callout } from "@/components/ui/callout";
import { Badge } from "@/components/ui/badge";
import { changelogData } from "@/data/changelog";

export const changeTypeBadge = {
  feature: { label: "Feature", variant: "info" },
  improvement: { label: "Improvement", variant: "updated" },
  fix: { label: "Fix", variant: "fix" },
  internal: { label: "Internal", variant: "secondary" },
};

export const formatVersion = (version) =>
  version.startsWith("v") ? version : `v${version}`;

<Callout title="Alpha Stage" type="warn">
  **Bilal UI** is currently in **Alpha**. Components are being developed and breaking changes may occur.
</Callout>

{changelogData.map((entry, index) => {
  const isAlpha = entry.version.toLowerCase().includes("alpha");

  return (
    <section key={`${entry.version}-${entry.date}`}>
      <h2>
        {formatVersion(entry.version)}{" "}
        {isAlpha ? (
          <Badge variant="alpha" className="ml-2 align-middle">
            Alpha
          </Badge>
        ) : (
          <Badge variant="version" className="ml-2 align-middle">
            Release
          </Badge>
        )}
      </h2>

      <p>
        <em>{entry.date}</em>
      </p>

      <p>{entry.description}</p>

      <ul>
        {entry.changes.map((change, changeIndex) => {
          const badge = changeTypeBadge[change.type] ?? {
            label: "Update",
            variant: "secondary",
          };

          return (
            <li key={`${entry.version}-${changeIndex}`}>
              <Badge
                variant={badge.variant}
                appearance="light"
                size="xs"
                className="mr-2 align-middle"
              >
                {badge.label}
              </Badge>
              {change.content}
            </li>
          );
        })}
      </ul>

      {index < changelogData.length - 1 ? <hr /> : null}
    </section>
  );
})}
